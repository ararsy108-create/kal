Solusi tunggal

Sistem persamaan linier yaitu kumpulan dari dua atau lebih persamaan linier dengan variabel yang sama. Tujuan utama: menemukan nilai variabel yang memenuhi semua persamaan secara bersamaan.

\(x + y = 5\)
\(2x - y = 1\)


                import numpy as np

                def solve_spl(matrix):
                    A = np.array(matrix, dtype=float)
                    n = len(A)

                    for i in range(n):
                        # Pivot
                        if A[i][i] == 0:
                            for j in range(i+1, n):
                                if A[j][i] != 0:
                                    A[[i, j]] = A[[j, i]]
                                    break

                        # Normalisasi pivot jadi 1
                        A[i] = A[i] / A[i][i]

                        # Eliminasi bawah
                        for j in range(i+1, n):
                            A[j] = A[j] - A[j][i] * A[i]

                    # Back substitution
                    x = np.zeros(n)
                    for i in range(n-1, -1, -1):
                        x[i] = A[i][-1] - sum(A[i][j] * x[j] for j in range(i+1, n))

                    return x

                # Contoh penggunaan
                matrix = np.array([[1, 1, 5],
                                [2, -1, 1]])

                hasil = solve_spl(matrix)
                print("x + y = 5\n2x - y = 1")
                print(f"bentuk matriksnya yaitu \n{matrix}")
                print("Solusi:", hasil)

Solusi Banyak

\[ \begin{cases} x + y + z = 6 \\ 2x + 2y + 2z = 12 \\ 3x + 3y + 3z = 18 \end{cases} \]


                import numpy as np

                def spl_solusi_banyak_dengan_langkah():
                    # Matriks augmented
                    A = np.array([
                        [1, 1, 1, 6],
                        [2, 2, 2, 12],
                        [3, 3, 3, 18]
                    ], dtype=float)

                    print("LANGKAH 0: Matriks Augmented Awal")
                    print(A)
                    print("-" * 40)

                    # =========================
                    # Langkah 1: R2 -> R2 - 2R1
                    # =========================
                    print("LANGKAH 1: R2 → R2 - 2R1")
                    A[1] = A[1] - 2 * A[0]
                    print(A)
                    print("-" * 40)

                    # =========================
                    # Langkah 2: R3 -> R3 - 3R1
                    # =========================
                    print("LANGKAH 2: R3 → R3 - 3R1")
                    A[2] = A[2] - 3 * A[0]
                    print(A)
                    print("-" * 40)

                    # =========================
                    # Kesimpulan
                    # =========================
                    print("KESIMPULAN:")
                    print("Terdapat dua baris nol.")
                    print("Artinya hanya ada satu persamaan independen.")
                    print("Sistem memiliki SOLUSI BANYAK (tak hingga).")
                    print()

                    print("Persamaan tersisa:")
                    print("x + y + z = 6")
                    print()

                    print("Misalkan:")
                    print("y = s")
                    print("z = t")
                    print()

                    print("Maka solusi umum:")
                    print("x = 6 - s - t")
                    print("y = s")
                    print("z = t")

                # Panggil fungsi
                spl_solusi_banyak_dengan_langkah()

                import numpy as np

                def spl_tidak_ada_solusi_dengan_langkah():
                    # Matriks augmented
                    A = np.array([
                        [1, 1, 1, 6],
                        [2, 2, 2, 12],
                        [1, 1, 1, 10]
                    ], dtype=float)

                    print("LANGKAH 0: Matriks Augmented Awal")
                    print(A)
                    print("-" * 50)

                    # =========================
                    # Langkah 1: R2 -> R2 - 2R1
                    # =========================
                    print("LANGKAH 1: R2 → R2 - 2R1")
                    A[1] = A[1] - 2 * A[0]
                    print(A)
                    print("-" * 50)

                    # =========================
                    # Langkah 2: R3 -> R3 - R1
                    # =========================
                    print("LANGKAH 2: R3 → R3 - R1")
                    A[2] = A[2] - A[0]
                    print(A)
                    print("-" * 50)

                    # =========================
                    # Kesimpulan
                    # =========================
                    print("KESIMPULAN:")
                    print("Muncul baris:")
                    print(A[2])
                    print()
                    print("Artinya:")
                    print("0x + 0y + 0z = 4")
                    print("0 = 4 (Kontradiksi)")
